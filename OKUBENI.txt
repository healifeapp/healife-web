bloguna ekleyin:

  - listen 80 -> return 301 https://hea-life.com$request_uri;
  - error_page 404 /404.html;
  - add_header satirlari: .htaccess icindeki Header always set ... degerleri
  - location ~* \.(css|js|webp|jpg|png|svg)$ { expires 1y; add_header
    Cache-Control "public, immutable"; }

    must-revalidate"; }
